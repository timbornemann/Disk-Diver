cmake_minimum_required(VERSION 3.22)
set(DISK_DIVER_VERSION "0.1.0" CACHE STRING "Disk-Diver release version (major.minor.patch)")
if(NOT DISK_DIVER_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "Expected a release version such as 1.2.3")
endif()
project(DiskDiver VERSION ${DISK_DIVER_VERSION} LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(DISK_DIVER_GUI "Build the desktop application" ON)
option(DISK_DIVER_NTFS_LAYOUT "Enable the benchmark-validated Windows volume layout backend" OFF)
set(DISK_DIVER_ENGINE "rust" CACHE STRING "Desktop scanner: rust or cpp")
set_property(CACHE DISK_DIVER_ENGINE PROPERTY STRINGS cpp rust)
if(NOT DISK_DIVER_ENGINE MATCHES "^(cpp|rust)$")
  message(FATAL_ERROR "DISK_DIVER_ENGINE must be cpp or rust")
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(cpp)
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:disk-diver-cpp>/disk-diver-engine.txt"
     CONTENT "${DISK_DIVER_ENGINE}\n")
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:disk-diver-cpp>/disk-diver-ntfs-layout.txt"
     CONTENT "${DISK_DIVER_NTFS_LAYOUT}\n")
install(FILES "$<TARGET_FILE_DIR:disk-diver-cpp>/disk-diver-engine.txt"
              "$<TARGET_FILE_DIR:disk-diver-cpp>/disk-diver-ntfs-layout.txt"
        DESTINATION ${CMAKE_INSTALL_BINDIR})
if(DISK_DIVER_ENGINE STREQUAL "rust")
  find_program(CARGO_EXECUTABLE cargo REQUIRED)
  set(rust_binary "${CMAKE_BINARY_DIR}/cargo/release/disk-diver-rust${CMAKE_EXECUTABLE_SUFFIX}")
  add_custom_target(disk-diver-rust ALL
    COMMAND "${CARGO_EXECUTABLE}" build --release --locked
            --manifest-path "${CMAKE_SOURCE_DIR}/rust/Cargo.toml"
            --target-dir "${CMAKE_BINARY_DIR}/cargo"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${rust_binary}"
            "$<TARGET_FILE_DIR:disk-diver-cpp>/disk-diver-rust${CMAKE_EXECUTABLE_SUFFIX}"
    VERBATIM)
  install(PROGRAMS "${rust_binary}" DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(DISK_DIVER_GUI)
  set(DISK_DIVER_UPDATE_REPOSITORY "" CACHE STRING "Public GitHub owner/repository for signed updates")
  set(DISK_DIVER_UPDATE_PUBLIC_KEY "" CACHE STRING "Base64 Ed25519 public key for release verification")
  set(default_update_platform "")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$" AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    if(WIN32)
      set(default_update_platform windows-x64)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
      set(default_update_platform linux-deb-amd64)
    endif()
  endif()
  set(DISK_DIVER_UPDATE_PLATFORM "${default_update_platform}" CACHE STRING "Update package platform")
  if(DISK_DIVER_UPDATE_REPOSITORY OR DISK_DIVER_UPDATE_PUBLIC_KEY)
    string(LENGTH "${DISK_DIVER_UPDATE_PUBLIC_KEY}" key_length)
    if(NOT DISK_DIVER_UPDATE_REPOSITORY MATCHES "^[a-zA-Z0-9-]+/[a-zA-Z0-9_.-]+$"
       OR DISK_DIVER_UPDATE_REPOSITORY MATCHES "/\\.\\.?$"
       OR NOT DISK_DIVER_UPDATE_PUBLIC_KEY MATCHES "^[a-zA-Z0-9+/]+=$" OR NOT key_length EQUAL 44)
      message(FATAL_ERROR "Configure both a GitHub repository and a base64-encoded Ed25519 public key")
    endif()
    if((WIN32 AND NOT DISK_DIVER_UPDATE_PLATFORM STREQUAL "windows-x64")
       OR (CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT DISK_DIVER_UPDATE_PLATFORM MATCHES "^linux-(deb-amd64|rpm-x86_64)$")
       OR NOT default_update_platform)
      message(FATAL_ERROR "Unsupported update platform for this target architecture")
    endif()
    set(updates_enabled 1)
  else()
    set(updates_enabled 0)
  endif()
  add_subdirectory(gui)
  configure_file(packaging/update-config.json.in disk-diver-update.json @ONLY)
  file(GENERATE OUTPUT "$<TARGET_FILE_DIR:disk-diver>/disk-diver-update.json"
       INPUT "${CMAKE_CURRENT_BINARY_DIR}/disk-diver-update.json")
  file(GENERATE OUTPUT "$<TARGET_FILE_DIR:disk-diver>/updater.py" INPUT "${CMAKE_CURRENT_SOURCE_DIR}/updater.py")
  install(FILES updater.py "${CMAKE_CURRENT_BINARY_DIR}/disk-diver-update.json" DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(TARGETS disk-diver RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    file(GENERATE OUTPUT "$<TARGET_FILE_DIR:disk-diver>/install_update.py" INPUT "${CMAKE_CURRENT_SOURCE_DIR}/install_update.py")
    install(FILES install_update.py DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES packaging/disk-diver.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES assets/disk-diver.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
    install(FILES assets/disk-diver.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
  endif()
endif()
install(TARGETS disk-diver-cpp RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES worker.py dd_journal.py dd_privilege.py DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE NOTICE DESTINATION ${CMAKE_INSTALL_DATADIR}/disk-diver)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME update-client COMMAND ${Python3_EXECUTABLE} "${PROJECT_SOURCE_DIR}/tests/test_updater.py")
  add_test(NAME update-install-stream COMMAND ${Python3_EXECUTABLE} "${PROJECT_SOURCE_DIR}/tests/test_install_update.py")
  add_test(NAME release-manifest COMMAND ${Python3_EXECUTABLE} -m unittest discover -s "${PROJECT_SOURCE_DIR}/tests" -p "test_release*.py")
  add_test(NAME cpp-cache-directory-identity COMMAND ${Python3_EXECUTABLE}
    "${PROJECT_SOURCE_DIR}/tests/check_excluded_dirs.py" "$<TARGET_FILE:disk-diver-cpp>")
  if(TARGET disk-diver-rust)
    add_test(NAME rust-cache-directory-identity COMMAND ${Python3_EXECUTABLE}
      "${PROJECT_SOURCE_DIR}/tests/check_excluded_dirs.py" "${rust_binary}")
  endif()
  add_test(NAME worker-integration
    COMMAND ${CMAKE_COMMAND} -E env "DISK_DIVER_BIN_DIR=$<TARGET_FILE_DIR:disk-diver-cpp>"
    ${Python3_EXECUTABLE} "${PROJECT_SOURCE_DIR}/tests/test_worker.py")
  add_test(NAME benchmark-selection
    COMMAND ${Python3_EXECUTABLE} -m unittest discover -s "${PROJECT_SOURCE_DIR}/bench" -p "test*.py")
  add_test(NAME journal-cursors
    COMMAND ${Python3_EXECUTABLE} "${PROJECT_SOURCE_DIR}/tests/test_journal.py")
endif()

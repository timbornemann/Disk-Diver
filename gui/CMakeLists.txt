find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
if(UNIX AND NOT APPLE)
    find_package(Qt6 6.2 REQUIRED COMPONENTS DBus)
endif()

add_library(disk-diver-ui STATIC window.cpp index.cpp treemap.cpp)
target_compile_features(disk-diver-ui PUBLIC cxx_std_17)
target_link_libraries(disk-diver-ui PUBLIC Qt6::Widgets)
add_executable(disk-diver WIN32 main.cpp)
set(icon_dir "${CMAKE_CURRENT_SOURCE_DIR}/../assets")
qt_add_resources(disk-diver "disk_diver_icons"
    PREFIX "/icons"
    BASE "${icon_dir}"
    FILES "${icon_dir}/disk-diver.png")
target_link_libraries(disk-diver PRIVATE disk-diver-ui)
if(WIN32)
    target_sources(disk-diver PRIVATE disk-diver.rc)
    target_include_directories(disk-diver PRIVATE "${icon_dir}")
    set_property(SOURCE disk-diver.rc PROPERTY OBJECT_DEPENDS "${icon_dir}/disk-diver.ico")
    target_link_libraries(disk-diver-ui PUBLIC shell32 ole32)
    if(MSVC)
        target_compile_options(disk-diver-ui PRIVATE /utf-8)
    endif()
elseif(UNIX)
    target_link_libraries(disk-diver-ui PRIVATE Qt6::DBus)
endif()
if(NOT DEFINED DISK_DIVER_WORKER_PATH)
    set(DISK_DIVER_WORKER_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../worker.py")
endif()
target_compile_definitions(disk-diver-ui PRIVATE DISK_DIVER_WORKER_PATH="${DISK_DIVER_WORKER_PATH}")
if(BUILD_TESTING)
    add_executable(disk-diver-ui-test integration_test.cpp)
    target_link_libraries(disk-diver-ui-test PRIVATE disk-diver-ui)
    add_dependencies(disk-diver-ui-test disk-diver-cpp)
    if(MSVC)
        target_compile_options(disk-diver-ui-test PRIVATE /utf-8)
    endif()
    add_test(NAME gui-scan-navigation COMMAND disk-diver-ui-test)
    set_tests_properties(gui-scan-navigation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 40)
endif()

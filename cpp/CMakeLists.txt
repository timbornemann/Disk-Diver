cmake_minimum_required(VERSION 3.20)
project(disk_diver_cpp LANGUAGES CXX)

add_executable(disk-diver-cpp main.cpp protocol.cpp)
target_compile_features(disk-diver-cpp PRIVATE cxx_std_20)
if(WIN32)
    target_sources(disk-diver-cpp PRIVATE windows_scan.cpp ntfs_layout.cpp)
    target_compile_definitions(disk-diver-cpp PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    if(MINGW)
        target_link_options(disk-diver-cpp PRIVATE -municode)
    endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(disk-diver-cpp PRIVATE linux_scan.cpp)
    target_compile_definitions(disk-diver-cpp PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
else()
    message(FATAL_ERROR "Disk-Diver supports Windows and Linux")
endif()
if(MSVC)
    target_compile_options(disk-diver-cpp PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(disk-diver-cpp PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING AND WIN32)
    add_executable(disk-diver-layout-test ntfs_layout_test.cpp ntfs_layout.cpp)
    target_compile_features(disk-diver-layout-test PRIVATE cxx_std_20)
    target_compile_definitions(disk-diver-layout-test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    add_test(NAME ntfs-layout-records COMMAND disk-diver-layout-test)
endif()
